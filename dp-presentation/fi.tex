%% SLIDE 4 — Analýza nedostatkov
\begin{frame}{Analýza nedostatkov}
\begin{block}{Technické problémy upstream}
  \begin{itemize}
    \item \alert{Nefunkčný FROST-Server} – kvôli nefunkčnému SMS backendu
    \item \alert{Upstream frontendy} – nevalidné MQTT správy
    \item Hardcoded UFZ konfigurácia, \textbf{žiadna dokumentácia}
    \item Nestabilný upstream vývoj – nekompatibilné zmeny,
          žiadna spätná kompatibilita
  \end{itemize}
\end{block}
\vspace{0.3em}
\begin{block}{Chýbajúce funkcie pre ČZU}
  \begin{itemize}
    \item Žiadny webový portál, projektová správa, správa senzorov
    \item Žiadne jednotné nasadenie, viacjazyčnosť, upozornenia
  \end{itemize}
\end{block}
\vspace{0.3em}
Celkovo \textbf{11 zasadných nedostatkov} $\rightarrow$ definujú rozsah práce.
\end{frame}
